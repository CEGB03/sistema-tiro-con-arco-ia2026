import os 
import json 
import tempfile from PIL 
import Image 
import numpy as np
import cv2 try: 
    import pypdfium2 as pdfium 
    PDF_SUPPORT = True
except ImportError: 
    PDF_SUPPORT = False
    

class AgenteClasificadorSegmentador:
    def __init__(self, output_dir=None):
        self.output_dir = output_dir or tempfile.mkdtemp(prefix="archery_agent1_")

    def analizar_layout(self, ruta_imagen):
        """
        Analiza la geometría y las líneas del documento para clasificar:
        - tipo_juego: 'clasificacion' o 'eliminatoria'
        - layout: '4_arqueros', '2_arqueros' o '1_arquero'
        - cuadrantes_activos: lista de identificadores con manuscritos
        """
        pil_img = Image.open(ruta_imagen)
        img_np = np.array(pil_img.convert('RGB'))
        gray = cv2.cvtColor(img_np, cv2.COLOR_RGB2GRAY)
        h, w = gray.shape

        filename = os.path.basename(ruta_imagen).lower()

        # Regla 1: Inspección de palabras clave en el nombre del archivo
        if "eliminacion" in filename or "duelo" in filename or "2-arqueros" in filename:
            tipo_juego = "eliminatoria"
            layout = "2_arqueros"
            cuadrantes = ["1A", "1B"]
        elif "clasificacion" in filename or "4-arqueros" in filename:
            tipo_juego = "clasificacion"
            layout = "4_arqueros"
            cuadrantes = ["1A", "1B", "1C", "1D"]
        else:
            # Regla 2: Análisis por Visión de Computadora (Morphology de líneas)
            _, thresh = cv2.threshold(gray, 200, 255, cv2.THRESH_BINARY_INV)
            kernel_h = cv2.getStructuringElement(cv2.MORPH_RECT, (max(10, w // 20), 1))
            lines_h = cv2.morphologyEx(thresh, cv2.MORPH_OPEN, kernel_h)

            top_density = np.mean(lines_h[0:int(h * 0.4), :])
            bot_density = np.mean(lines_h[int(h * 0.5):int(h * 0.9), :])

            if top_density &gt; 0.05 and bot_density &gt; 0.05:
                tipo_juego = "clasificacion"
                layout = "4_arqueros"
                cuadrantes = ["1A", "1B", "1C", "1D"]
            else:
                tipo_juego = "clasificacion"
                layout = "1_arquero"
                cuadrantes = ["Individual"]

        # Regla 3: Descarte de cuadrantes en blanco por densidad de tinta
        activos = []
        if layout == "4_arqueros":
            coords = {
                "1A": (0, 0, w // 2, h // 2),
                "1B": (w // 2, 0, w, h // 2),
                "1C": (0, h // 2, w // 2, h),
                "1D": (w // 2, h // 2, w, h)
            }
            _, thresh = cv2.threshold(gray, 200, 255, cv2.THRESH_BINARY_INV)
            for q_name, (x1, y1, x2, y2) in coords.items():
                crop = thresh[y1:y2, x1:x2]
                my, mx = int((y2 - y1) * 0.15), int((x2 - x1) * 0.15)
                inner = crop[my:-my, mx:-mx]
                if np.mean(inner) &gt; 0.8:
                    activos.append(q_name)
            if not activos:
                activos = cuadrantes
        else:
            activos = cuadrantes

        return tipo_juego, layout, activos
    

def procesar_documento(self, ruta_archivo):
        """
        Procesa el archivo de entrada (PDF o Imagen) y retorna un
        manifiesto JSON con las imágenes recortadas listas para el Agente 2.
        """
        es_pdf = ruta_archivo.lower().endswith('.pdf')
        paginas_imagenes = []

        if es_pdf:
            if not PDF_SUPPORT:
                raise ImportError("Se requiere 'pypdfium2' para procesar archivos PDF.")
            doc = pdfium.PdfDocument(ruta_archivo)
            for i, page in enumerate(doc):
                bitmap = page.render(scale=2)
                pil_img = bitmap.to_pil()
                img_path = os.path.join(self.output_dir, f"pagina_{i+1}.png")
                pil_img.save(img_path)
                paginas_imagenes.append((i + 1, img_path))
        else:
            paginas_imagenes.append((1, ruta_archivo))

        manifiesto = {
            "archivo_origen": os.path.basename(ruta_archivo),
            "total_paginas": len(paginas_imagenes),
            "tareas_agente2": []
        }

        for num_pag, img_path in paginas_imagenes:
            tipo_juego, layout, activos = self.analizar_layout(img_path)
            img = Image.open(img_path)
            w, h = img.size

            if layout == "4_arqueros":
                cuadrantes_coords = {
                    "1A": (0, 0, w // 2, h // 2),
                    "1B": (w // 2, 0, w, h // 2),
                    "1C": (0, h // 2, w // 2, h),
                    "1D": (w // 2, h // 2, w, h)
                }
            elif layout == "2_arqueros":
                cuadrantes_coords = {
                    "1A": (0, 0, w // 2, h),
                    "1B": (w // 2, 0, w, h)
                }
            else:
                cuadrantes_coords = {"Individual": (0, 0, w, h)}

            for target_id in activos:
                coords = cuadrantes_coords[target_id]
                crop_img = img.crop(coords)
                crop_filename = f"pag_{num_pag}_crop_{target_id}.png"
                crop_path = os.path.join(self.output_dir, crop_filename)
                crop_img.save(crop_path, "PNG")

                manifiesto["tareas_agente2"].append({
                    "pagina": num_pag,
                    "target_id": target_id,
                    "tipo_juego": tipo_juego,
                    "ruta_recorte": crop_path
                })

        return manifiesto
